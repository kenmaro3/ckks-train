def digest(p):
    h=hashlib.sha256()
    with Path(p).open('rb') as f:
        for chunk in iter(lambda:f.read(1024*1024),b''):h.update(chunk)
    return h.hexdigest()
def manifest(path, base, commit=False):

            assert hashlib.sha256(original).hexdigest()==sha,(name,'commit mismatch')
        rows.append((name,sha))
    assert len({n for n,h in rows})==len(rows)
    return rows
